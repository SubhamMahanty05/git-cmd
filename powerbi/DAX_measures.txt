Inventory Value = SUM(inventory_analysis[inventory_value])
Products at Reorder Risk =

    inventory_analysis[stock_status] IN {"Stockout","Reorder Risk"})
Excess Stock Units =
CALCULATE(SUM(inventory_analysis[stock_quantity]),
    inventory_analysis[stock_status] = "Excess Stock")
Forecast Demand 30D = SUM(demand_forecast_30d[forecast_demand])
Recommended Replenishment Units = SUM(inventory_analysis[recommended_order_qty])
